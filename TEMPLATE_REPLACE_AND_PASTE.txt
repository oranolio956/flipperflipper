sudo apt install -y build-essential wget curl git vim ufw fail2ban python3 python3-pip python3-venv python3-dev libpq-dev libjpeg-dev zlib1g-dev libffi-dev libssl-dev libsqlite3-dev nginx certbot python3-certbot-nginx htop

mkdir -p /opt/elite-rat/data

# ========== SECTION 10: SSL CERTIFICATE ==========
# OPTION A: If you have a real domain name, use Let's Encrypt:
sudo certbot certonly --standalone -d YOUR_DOMAIN -d www.YOUR_DOMAIN

# OPTION B: If using just IP address, use self-signed certificate:
sudo openssl req -x509 -nodes -days 365 -newkey rsa:4096 -keyout /opt/elite-rat/ssl/key.pem -out /opt/elite-rat/ssl/cert.pem -subj "/C=US/ST=State/L=City/O=Org/CN=YOUR_DOMAIN"
sudo chown -R $USER:$USER /opt/elite-rat/ssl
chmod 600 /opt/elite-rat/ssl/key.pem


# ========== SECTION 11: CONFIGURE NGINX ==========
# This opens an editor:
sudo nano /etc/nginx/sites-available/elite-rat

# ↓↓↓ COPY EVERYTHING BETWEEN THE LINES AND PASTE INTO EDITOR ↓↓↓
# ===================================================================
upstream flask_app {
    server 127.0.0.1:5000;
}

server {
    listen 80;
    listen [::]:80;
    server_name YOUR_DOMAIN www.YOUR_DOMAIN;
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name YOUR_DOMAIN www.YOUR_DOMAIN;

    # If you used Let's Encrypt (Option A in Section 10):
    ssl_certificate /etc/letsencrypt/live/YOUR_DOMAIN/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/YOUR_DOMAIN/privkey.pem;
    
    # If you used self-signed (Option B in Section 10), comment above and uncomment below:
    # ssl_certificate /opt/elite-rat/ssl/cert.pem;
    # ssl_certificate_key /opt/elite-rat/ssl/key.pem;

    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;
    ssl_prefer_server_ciphers on;
    client_max_body_size 100M;

    location / {
        proxy_pass http://flask_app;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_connect_timeout 600s;
        proxy_send_timeout 600s;
        proxy_read_timeout 600s;
    }

    location /socket.io/ {
        proxy_pass http://flask_app/socket.io/;
        proxy_http_version 1.1;
        proxy_buffering off;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "Upgrade";
        proxy_set_header Host $host;
    }

    access_log /var/log/nginx/elite-rat-access.log;
    error_log /var/log/nginx/elite-rat-error.log;
}
# ===================================================================
# ↑↑↑ AFTER PASTING:
# 1. Replace ALL instances of YOUR_DOMAIN with your actual domain (or IP)
# 2. Choose the correct SSL certificate path (Let's Encrypt or self-signed)
# 3. Save: Press Ctrl+X, then Y, then Enter


# ========== SECTION 12: ENABLE NGINX ==========
sudo ln -s /etc/nginx/sites-available/elite-rat /etc/nginx/sites-enabled/
sudo rm -f /etc/nginx/sites-enabled/default
sudo nginx -t
sudo systemctl restart nginx
sudo systemctl enable nginx


# ========== SECTION 13: CREATE SYSTEMD SERVICE ==========
# This opens an editor:
sudo nano /etc/systemd/system/elite-rat.service

# ↓↓↓ COPY EVERYTHING BETWEEN THE LINES AND PASTE INTO EDITOR ↓↓↓
# ===================================================================
[Unit]
Description=Elite RAT Web Application
After=network.target

[Service]
Type=simple
User=root
Group=root
WorkingDirectory=/opt/elite-rat
Environment="PATH=/opt/elite-rat/venv/bin"
Environment="FLASK_ENV=production"
EnvironmentFile=/opt/elite-rat/.env

ExecStart=/opt/elite-rat/venv/bin/gunicorn \
    --bind 127.0.0.1:5000 \
    --workers 4 \
    --worker-class gevent \
    --timeout 300 \
    --access-logfile /opt/elite-rat/logs/access.log \
    --error-logfile /opt/elite-rat/logs/error.log \
    --log-level info \
    web_app_real:app

Restart=always
RestartSec=10
NoNewPrivileges=true
PrivateTmp=true

[Install]
WantedBy=multi-user.target
# ===================================================================
# ↑↑↑ Save: Press Ctrl+X, then Y, then Enter


# ========== SECTION 14: START THE SERVICE ==========
sudo systemctl daemon-reload
sudo systemctl enable elite-rat
sudo systemctl start elite-rat


# ========== SECTION 15: CHECK STATUS ==========
sudo systemctl status elite-rat
# ↑ Should show "active (running)" in green
# Press 'q' to exit


# ========== SECTION 16: SETUP FAIL2BAN ==========
sudo cp /etc/fail2ban/jail.conf /etc/fail2ban/jail.local
sudo systemctl enable fail2ban
sudo systemctl start fail2ban


# ========== SECTION 17: SETUP AUTOMATED BACKUPS ==========
sudo mkdir -p /opt/backups
cat > /opt/elite-rat/backup.sh << 'EOF'
#!/bin/bash
BACKUP_DIR="/opt/backups"
DATE=$(date +%Y%m%d_%H%M%S)
tar -czf $BACKUP_DIR/backup-$DATE.tar.gz /opt/elite-rat/data /opt/elite-rat/.env 2>/dev/null
ls -t $BACKUP_DIR/backup-*.tar.gz | tail -n +8 | xargs rm -f 2>/dev/null
echo "Backup complete: $BACKUP_DIR/backup-$DATE.tar.gz"
EOF
chmod +x /opt/elite-rat/backup.sh
(crontab -l 2>/dev/null; echo "0 2 * * * /opt/elite-rat/backup.sh") | crontab -


# ========== SECTION 18: VERIFY EVERYTHING ==========
echo "=== Checking Services ==="
sudo systemctl status elite-rat nginx fail2ban | grep Active

echo ""
echo "=== Checking Ports ==="
sudo netstat -tulpn | grep -E '(80|443|5000)'

echo ""
echo "=== Testing Local Connection ==="
curl -I http://localhost:5000

echo ""
echo "=== Testing External Connection ==="
# Replace YOUR_DOMAIN below:
curl -I https://YOUR_DOMAIN


# ========== SECTION 19: REMOVE TEMPORARY PORT ==========
sudo ufw delete allow 5000/tcp
sudo ufw reload
